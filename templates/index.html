let partnerDataCache = []; let localSavedIds = new Set(); // Store local student IDs to check for duplicates document.addEventListener('DOMContentLoaded', loadLocalDatabase); // 1. Load Local Database & Track Existing Student IDs function loadLocalDatabase() { fetch('/api/students') .then(res => res.json()) .then(data => { const localBody = document.getElementById('local-records-body'); const importedBody = document.getElementById('imported-records-body'); localBody.innerHTML = ''; importedBody.innerHTML = ''; // Clear and rebuild set of saved IDs localSavedIds.clear(); data.forEach(student => { localSavedIds.add(student.student_id); if (student.source === 'Local') { localBody.innerHTML += ` ${student.id} ${student.student_id} ${student.name} ${student.program} ${student.year} ${student.section} ${student.source} `; } else { importedBody.innerHTML += ` ${student.id} ${student.student_id} ${student.name} ${student.program} ${student.year} ${student.section} ${student.source} `; } }); // Re-render partner table if partner data is active if (partnerDataCache.length > 0) { renderPartnerTable(); } }); } // 2. Fetch Partner Records function fetchPartnerRecords() { const url = document.getElementById('partner-url-input').value.trim(); if (!url) return alert("Please enter partner URL"); fetch(`/partner-records?url=${encodeURIComponent(url)}`) .then(res => res.json()) .then(data => { if (data.error) throw new Error(data.error); partnerDataCache = data; renderPartnerTable(); }) .catch(err => alert("Error: " + err.message)); } // 3. Render Partner Table with Saved Button Logic function renderPartnerTable() { const tbody = document.getElementById('partner-records-body'); tbody.innerHTML = ''; partnerDataCache.forEach((student, index) => { const isAlreadySaved = localSavedIds.has(student.student_id); tbody.innerHTML += ` ${student.student_id} ${student.name} ${student.program} ${student.year} ${student.section} ${isAlreadySaved ? `` : `` } `; }); } // 4. Save Selected Partner Record with UI State Update function importPartnerRecord(index, btnElement) { const student = partnerDataCache[index]; // Show visual loading state immediately btnElement.innerText = "Saving..."; btnElement.disabled = true; fetch('/import-student', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(student) }) .then(res => res.json()) .then(data => { // Change button to Saved permanently btnElement.innerText = "Saved ✓"; btnElement.className = "btn btn-saved"; btnElement.disabled = true; // Refresh local tables loadLocalDatabase(); }) .catch(err => { alert("Failed to save record: " + err); btnElement.innerText = "Save to My Database"; btnElement.disabled = false; }); }
